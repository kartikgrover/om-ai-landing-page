# Om.AI Robots.txt
# Updated: Aug 9, 2026

User-agent: *
Allow: /

# Disallow non-public directories
Disallow: /node_modules/
Disallow: /blog/drafts/

# Backend API paths (referenced from inline JS, not site routes)
Disallow: /daily-horoscope/v2/
Disallow: /daily-verse

# Query-string URLs are deliberately NOT disallowed. A previous "Disallow: /*?"
# also matched the cache-busted stylesheet (/styles/styles.min.css?v=NN) on every
# page, so Googlebot rendered the whole site unstyled; and it kept the AppAgg
# homepage backlink (?from=AppAgg...) uncrawlable, so its equity never
# consolidated. Every indexable page carries rel=canonical — that handles
# parameter dedup without robots rules.

# Disallow certain file types
Disallow: /*.pdf$
Disallow: /*.zip$

# Sitemap location
Sitemap: https://omai.app/sitemap.xml
